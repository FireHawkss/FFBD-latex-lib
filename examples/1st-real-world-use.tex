% !TeX program = lualatex
\documentclass[a4paper]{article}
\usepackage[landscape,margin=16mm]{geometry}
\usepackage{tikzffbd}
\pagestyle{empty}

\begin{document}
\begin{ffbd}[
    theme=slate,
    warp=true,
    max-columns=8,
    annotations=below
]

\start{start}{Drone state = deployed}
\function{init-scan}{Conduct disaster area flyover scan}
\function{relay}{Relay findings to HQ}
\function{choose-location}{Choose a location to search}
\function{move-to-loc}{Move to search location}
\function{search-location}{Search the location}
% \function{relay-info}{Relay information to HQ}
\function{mission-status}{Evaulate mission status}
\function{continue}{Continue mission}
\finish{end}{Return to base}


% Initial assessment
\flow{start}{init-scan}
\flow{init-scan}{relay}


% Search cycle
\flow{relay}{choose-location}
\flow{choose-location}{move-to-loc}
\flow{move-to-loc}{search-location}
\flow{search-location}{mission-status}

% Decide whether to continue searching or return
\branch[or]{mission-status}{continue,end}

% If continuing, determine the next location and repeat
\loopflow[side=above]{continue}{choose-location}





\end{ffbd}
\end{document}
